def imagen_uri(nombre):
    """Convierte una imagen de la carpeta 'imagenes' en data-URI para usarla en HTML/CSS."""
    ruta = CARPETA_IMAGENES / nombre
    if not ruta.exists():
        return ""
    mime = "image/jpeg" if ruta.suffix.lower() in (".jpg", ".jpeg") else "image/png"
    datos = base64.b64encode(ruta.read_bytes()).decode()
    return f"data:{mime};base64,{datos}"

def identificar_prenda(ruta_imagen):
    tipo = random.choice(TIPOS)
    color = "Por detectar"
    return tipo, color

def nombre_header(nombre, usuario):
    nombre = nombre or "Tu nombre"
    usuario = usuario or "@usuario"
    return f"<div class='nombre'>{nombre} ✦<br><small>{usuario}</small></div>"


def iniciar_sesion(nombre, usuario):
    nombre = (nombre or "").strip()
    usuario = (usuario or "").strip().lstrip("@")
    if not nombre or not usuario:
        return (
            gr.update(), gr.update(),
            "<div class='aviso'>⚠️ Escribe tu nombre y tu usuario.</div>",
            gr.update(), gr.update(), gr.update(),
        )
    usuario = "@" + usuario.replace(" ", "_").lower()
    return (
        gr.update(visible=False),   # oculta la portada
        gr.update(visible=True),    # muestra la app
        "",
        nombre_header(nombre, usuario),
        nombre,
        usuario,
    )


def cerrar_sesion():
    return gr.update(visible=True), gr.update(visible=False)

IDS_PESTANAS = ["escanear", "closet", "perfil"]


def ir_a(destino):
    return (gr.update(selected=destino),) + tuple(
        gr.update(variant="primary" if i == destino else "secondary") for i in IDS_PESTANAS
    )


# ---------------------------------------------------------------------
#  Funciones de la app
# ---------------------------------------------------------------------
def escanear(imagen):
    if imagen is None:
        return "<div class='aviso'>📷 Toma o sube una foto primero.</div>", None
    tipo, color = identificar_prenda(imagen)
    tarjeta = f"""
    <div class='tarjeta'>
      <div class='check'>✓</div>
      <h3>{tipo}</h3>
      <p>Color: {color}</p>
    </div>"""
    return tarjeta, {"imagen": imagen, "tipo": tipo, "color": color}


def ver_closet(filtro):
    items = []
    for p in closet:
        if filtro == "Todas" or PLURAL[p["tipo"]] == filtro:
            items.append((p["imagen"], f'{p["tipo"]} · {p["color"]}'))
    return items


def stats_html():
    n = len(closet)
    c = sum(1 for p in closet if p["tipo"] == "Chaqueta")
    pa = sum(1 for p in closet if p["tipo"] == "Pantalón")
    ca = sum(1 for p in closet if p["tipo"] == "Camisa")
    return f"""
    <div class='stats'>
      <div><b>{n}</b><span>Prendas</span></div>
      <div><b>{c}</b><span>Chaquetas</span></div>
      <div><b>{pa}</b><span>Pantalones</span></div>
      <div><b>{ca}</b><span>Camisas</span></div>
    </div>"""


def guardar(detectada):
    if detectada is None:
        return "⚠️ Primero escanea una prenda.", ver_closet("Todas"), stats_html()
    closet.append(detectada)
    return (
        f"✅ {detectada['tipo']} guardada en tu closet.",
        ver_closet("Todas"),
        stats_html(),
    )

def _capas(clase, capas):
    """Genera el CSS de un fondo con varias imágenes: [(nombre, posición, tamaño), ...]"""
    imgs = ", ".join(f"var(--img-{n})" for n, _, _ in capas)
    pos = ", ".join(p for _, p, _ in capas)
    tam = ", ".join(t for _, _, t in capas)
    return (
        f".{clase} {{ background-image: {imgs} !important; "
        f"background-position: {pos} !important; background-size: {tam} !important; "
        f"background-repeat: no-repeat !important; }}"
    )
